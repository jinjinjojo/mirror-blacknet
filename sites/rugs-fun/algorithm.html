generatePrices() — fully recovered

Lifted verbatim from generatePrices-Cb6Po-5f.js (16 LoC post-unminify). Given (serverSeed, roundId, externalHash?) it emits the entire price array including the exact rug tick.

function s(serverSeed, roundId, externalHash) {
  const rng = seedrandom(
    externalHash
      ? `${serverSeed}|${externalHash}`              // Full Shield
      : isUuidV7with7at14(roundId)
        ? serverSeed                                   // "7-at-14" shortcut (!!)
        : `${serverSeed}-${roundId}`                  // Standard
  );
  let prices = [1], u = 1, d = 1;
  for (let i = 0; i < 2**53 - 1; i++) {
    if (rng() < 0.005 || u <= 0) {                     // RUG
      u = u <= 0 ? 0 : d * (1 - 0.98);                 // drops to 2 % of max
      prices.push(u);
      break;
    }
    if (rng() < 1e-5 && u <= 100) { u *= 10; }         // GOD CANDLE
    else if (rng() < 0.125) {                          // BIG MOVE
      const mag = 0.15 + rng() * 0.10;
      u *= 1 + (rng() > 0.5 ? mag : -mag);
    } else {                                           // DRIFT
      const drift = -0.02 + rng() * 0.05 + 0.005 * Math.min(10, Math.sqrt(u)) * (2*rng() - 1);
      u *= 1 + drift;
    }
    if (u < 0) u = 0;
    if (u < d) d = u;
    prices.push(u);
  }
  return prices;
}
